cmake_minimum_required(VERSION 3.20)
project(minivectordb LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mvdb_core src/storage.cpp src/sql.cpp src/simd.cpp src/engine.cpp)
find_package(ZLIB REQUIRED)
target_link_libraries(mvdb_core PUBLIC ZLIB::ZLIB)
target_compile_options(mvdb_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mvdb_local src/local_adapter.cpp)
target_link_libraries(mvdb_local PRIVATE mvdb_core)
target_compile_options(mvdb_local PRIVATE -Wall -Wextra -Wpedantic)
add_executable(benchmark_cpu EXCLUDE_FROM_ALL tools/benchmark_cpu.cpp)
target_include_directories(benchmark_cpu PRIVATE src)
target_link_libraries(benchmark_cpu PRIVATE mvdb_core)
enable_testing()
add_executable(budget_regression tests/budget_regression.cpp)
target_include_directories(budget_regression PRIVATE src)
target_compile_options(budget_regression PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_test(NAME budget_regression COMMAND budget_regression)
add_executable(storage_regression tests/storage_regression.cpp)
target_link_libraries(storage_regression PRIVATE mvdb_core)
target_compile_options(storage_regression PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_test(NAME storage_regression COMMAND storage_regression)
add_executable(core_regression tests/core_regression.cpp tests/sql_stream_regression.cpp tests/r12_engine_regression.cpp tests/r12_query_regression.cpp)
target_link_libraries(core_regression PRIVATE mvdb_core)
target_compile_options(core_regression PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_test(NAME core_regression COMMAND core_regression)
add_executable(sql_vector_benchmark EXCLUDE_FROM_ALL tests/sql_stream_regression.cpp src/sql.cpp)
target_compile_definitions(sql_vector_benchmark PRIVATE MVDB_SQL_STREAM_STANDALONE)
target_compile_options(sql_vector_benchmark PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_executable(simd_regression tests/simd_regression.cpp)
target_link_libraries(simd_regression PRIVATE mvdb_core)
target_compile_options(simd_regression PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_test(NAME simd_regression COMMAND simd_regression)
add_executable(index_regression tests/index_regression.cpp tests/r12_index_regression.cpp)
target_link_libraries(index_regression PRIVATE mvdb_core)
target_compile_options(index_regression PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_test(NAME index_regression COMMAND index_regression)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(adapter_serializer_regression tests/adapter_serializer_regression.cpp)
target_link_libraries(adapter_serializer_regression PRIVATE mvdb_core)
target_compile_options(adapter_serializer_regression PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_test(NAME protocol_regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/protocol_regression.py $<TARGET_FILE:mvdb_local> $<TARGET_FILE:adapter_serializer_regression>)
add_test(NAME all COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/verify.py --binary $<TARGET_FILE:mvdb_local> --repo ${CMAKE_SOURCE_DIR})

execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy" RESULT_VARIABLE MVDB_NUMPY_STATUS OUTPUT_QUIET ERROR_QUIET)
if(MVDB_NUMPY_STATUS EQUAL 0)
  add_test(NAME dynamic_dml COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/dynamic_dml.py --binary $<TARGET_FILE:mvdb_local> --output ${CMAKE_BINARY_DIR}/dynamic-dml.json)
  add_test(NAME low_memory_regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/low_memory.py
           --binary $<TARGET_FILE:mvdb_local> --phase ctest --budgets 4194304 8388608 10485760 --no-default-budget --output ${CMAKE_BINARY_DIR}/low-memory.json --enforce-latency --require-ivf)
endif()
